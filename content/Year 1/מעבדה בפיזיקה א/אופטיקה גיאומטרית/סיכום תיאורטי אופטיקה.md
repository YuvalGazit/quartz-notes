$$
\text{(1)  } \frac{1}{f}=\frac{1}{v}+\frac{1}{u}
$$

$$
\text{(2) }M_{T}=\frac{h}{H}=-\frac{v}{u}
$$

$$
\begin{matrix}
u=\infty \\
\frac{1}{f}=\frac{1}{v}+0 \\
f=v
\end{matrix}
$$
