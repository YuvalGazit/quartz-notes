\vec{r}_{AB}=\vec{r}_{B}-\vec{r}_{A}=(x_{B}-x_{A},y_{B}-y_{A},d) \\

\end{matrix}
$$
## ב
$$
\begin{matrix}
\vec{r}_{AB}\times \vec{r}_{BC}= \\
\hat{x}((y_{B}-y_{A})*0-d(y_{A}-y_{B})) \\
-\hat{y}((x_{B}-x_{A})*0-d(x_{A}-x_{B})) \\
+\hat{z}((x_{B}-x_{A})(y_{A}-y_{B})-(y_{B}-y_{A})(x_{A}-x_{B})) \\
= \\
\hat{x}(-d(y_{A}-y_{B})) \\
-\hat{y}(-d(x_{A}-x_{B})) \\
+\hat{z}(\cancel{ (x_{B}y_{A}-x_{B}y_{B}-x_{A}y_{A}+x_{A}y_{B}) } -\cancel{ (y_{B}x_{A}-y_{B}x_{B}-y_{A}x_{A}+y_{A}x_{B}) })  \\
=-d\hat{x}(y_{A}-y_{B})+d\hat{y}(x_{A}-x_{B})+\hat{z}*0 \\
=d(-\hat{x}(y_{A}-y_{B})+\hat{y}(x_{A}-x_{B}))
\end{matrix}
$$
המרכיבים הם אותם מרכיבים של $\vec{r}_{BC}$ , רק כפול d - ולכן הגודל יהיה כמו של $\vec{r}_{BC}$ אבל כפול d - כלומר $|\vec{r}_{AB}\times \vec{r}_{BC}|=d\cdot|\vec{r}_{BC}|$

# 5
$$
\begin{cases}
\vec{r}_{A}(t)=\vec{r}_{A}(t_{0})+\vec{v}_{A}t 
=(a_{1},0,a_{3})+(V_{A}t,0,0) & =(V_{A}t+a_{1},0,a_{3}) \\
\vec{r}_{B}(t)=\vec{r}_{B}(t_{0})+\vec{v}_{B}t 
=(0,b_{2},b_{3})+(V_{B}t,V_{B}t,0) & =(V_{B}t,V_{B}t+b_{2},b_{3})
\end{cases}
$$
המרחק בין הלווינים מתבטא על ידי גודל הוקטור שיוצא מחיסור וקטורי המיקום. נשאיר את המרחק בריבוע כדי להפשיט את המשוואה, וכי המרחק בריבטע המינימלי עדיין יהיה המרחק המינימלי, רק בריבוע.
$$
|\vec{d}(t)|^{2}=|\vec{r}_{A}(t)-\vec{r}_{B}(t)|^{2}
= (V_{A}t+a_{1}-V_{B}t)^{2}+(0-(V_{B}t+b_{2}))^{2}+(a_{3}-b_{3})^{2} 
$$
$$
\begin{matrix}
=V_{A}^{2}t^{2}+2a_{1}V_{A}t-2V_{A}V_{B}t^{2}+a_{1}^{2}-2a_{1}V_{B}t+\cancel{ V_{B}t^{2} } \\
\cancel{ -V_{B}t^{2} }-2V_{B}tb_{2}-b_{2}^{2} \\
+a_{3}^{2}-2a_{3}b_{3}+b_{3}^{2}
\end{matrix}
$$
$$
=V_{A}^{2}t^{2}+2a_{1}V_{A}t-2V_{A}V_{B}t^{2}+a_{1}^{2}-2a_{1}V_{B}t-2V_{B}tb_{2}-b_{2}^{2}
+a_{3}^{2}-2a_{3}b_{3}+b_{3}^{2}$$
$$
=t^{2}(V_{A}^{2}-2V_{A}V_{B})
+2t(a_{1}V_{A}-a_{1}V_{B}-V_{B}b_{2})
+(a_{1}^{2}-b_{2}^{2}+a_{3}^{2}-2a_{3}b_{3}+b_{3}^{2})
$$

נגזור את התוצאה $|\vec{d}(t)|^{2}$ כמו פונקציה ונשווה לאפס, כדי למצוא נקודות מינימום ומקסימום.
$$
|\vec{d}| ^{2} (t)'=\cancel{ 2 }t(V_{A}^{2}-2V_{A}V_{B})
+\cancel{ 2 }(a_{1}V_{A}-a_{1}V_{B}-V_{B}b_{2})=0
$$
$$
t=-\frac{a_{1}V_{A}-a_{1}V_{B}-V_{B}b_{2}}{V_{A}^{2}-2V_{A}V_{B}}
$$
נציב את הזמן שקיבלנו חזרה בפונקצייה כדי למצוא את המרחק בריבוע שמתקיים בזמן הזה.
$$
\begin{matrix}
d_{min}^{2}=\left( -\frac{a_{1}V_{A}-a_{1}V_{B}-V_{B}b_{2}}{V_{A}^{2}-2V_{A}V_{B}} \right)^{2}(V_{A}^{2}-2V_{A}V_{B})  \\
+2 \left( -\frac{a_{1}V_{A}-a_{1}V_{B}-V_{B}b_{2}}{V_{A}^{2}-2V_{A}V_{B}} \right) (a_{1}V_{A}-a_{1}V_{B}-V_{B}b_{2}) \\
+(a_{1}^{2}-b_{2}^{2}+a_{3}^{2}-2a_{3}b_{3}+b_{3}^{2})
\end{matrix}
$$
$$
\begin{matrix}
=\frac{(a_{1}V_{A}-a_{1}V_{B}-V_{B}b_{2})^{2}}{(V_{A}^{2}-2V_{A}V_{B})\cancel{ ^{2} }}\cancel{ (V_{A}^{2}-2V_{A}V_{B}) }  \\
-2 \frac{(a_{1}V_{A}-a_{1}V_{B}-V_{B}b_{2})^{2}}{V_{A}^{2}-2V_{A}V_{B}} \\
+(a_{1}^{2}-b_{2}^{2}+a_{3}^{2}-2a_{3}b_{3}+b_{3}^{2})
\end{matrix}
$$
$$
=- \frac{(a_{1}V_{A}-a_{1}V_{B}-V_{B}b_{2})^{2}}{V_{A}^{2}-2V_{A}V_{B}} 
+(a_{1}^{2}-b_{2}^{2}+a_{3}^{2}-2a_{3}b_{3}+b_{3}^{2})
$$
כלומר
$$
d_{min}=\sqrt{ - \frac{(a_{1}V_{A}-a_{1}V_{B}-V_{B}b_{2})^{2}}{V_{A}^{2}-2V_{A}V_{B}} 
+a_{1}^{2}-b_{2}^{2}+a_{3}^{2}-2a_{3}b_{3}+b_{3}^{2} }
$$
