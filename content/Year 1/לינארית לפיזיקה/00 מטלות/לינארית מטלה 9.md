$$
A=\begin{pmatrix}
a & 0 & a \\
1 & 1 & 1 \\
3 & 2 & a
\end{pmatrix}
$$
$$
|A|=\begin{vmatrix}
a & 0 & a \\
1 & 1 & 1 \\
3 & 2 & a
\end{vmatrix}=\begin{vmatrix}
-a & 0 & a \\
0 & 0 & 1 \\
4 & -3+a & 0
\end{vmatrix}=-1\begin{vmatrix}
-a & 0 \\
4 & -3+a
\end{vmatrix}=-3a+a^{2}
$$

a\neq 0 & a\neq 3

## א
$$
p(\lambda)=|A-\lambda I|=\begin{vmatrix}
2-\lambda & 0 & 0 \\
0 & 1-\lambda & -1 \\
0 & -1 & 1-\lambda
\end{vmatrix}=(2-\lambda)\begin{vmatrix}
1-\lambda & -1 \\
-1 & 1-\lambda
\end{vmatrix}
$$
$$
=(2-\lambda)[(1-\lambda)^{2}-1]=(2-\lambda)[-2\lambda+\lambda^{2}]=-4\lambda +2\lambda^{2}+2\lambda^{2} -\lambda^{3}
$$
$$
=-\lambda(4-4\lambda+\lambda^{2})=-\lambda(\lambda-2)^{2}
$$
