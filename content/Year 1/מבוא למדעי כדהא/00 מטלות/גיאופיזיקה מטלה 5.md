$$
\begin{cases}

\cos c=\cos 74\cos a+\sin 74\sin a\cos 80 & \implies & \cos c=0.27\cos a+0.17\sin a \\
\cos a=\cos 74\cos c+\sin 74\sin c\cos A & \implies & \cos A=\frac{\cos a-0.27\cos c}{0.96\sin c}=\frac{0.93\cos a-0.046\sin a}{0.96\sin \cos ^{-1}(0.27\cos a+0.17\sin a)} \\
\end{cases}
