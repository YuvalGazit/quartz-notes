m_{1}=\frac{1}{V_{1}} \implies \boxed{ V_{1}=1.5 \text{ m/ms} }
$$
$$
\begin{matrix}
t_{iu}=20 & t_{id}=40 \\
V_{u}=3 & V_{d}=4
\end{matrix}

$$
\theta=\frac{1}{2}\left[ \sin ^{-1} \frac{V_{1}}{V_{d}}-\sin ^{-1} \frac{V_{1}}{V_{u}} \right]= \boxed{ -3.99 \degree }
$$
מכיוון ש $\cos\theta=0.99 \approx1$, אז 
$$
\frac{1}{V_{2}}\approx \frac{1}{2}\left( \frac{1}{V_{d}}+\frac{1}{V_{u}} \right) \implies \boxed{ V_{2}= 3 \frac{3}{7}\text{ m/ms} }
$$
$$
i_{c}=\frac{1}{2}\left[ \sin ^{-1} \frac{V_{1}}{V_{d}}+\sin ^{-1} \frac{V_{1}}{V_{u}} \right]=26 \degree
$$
$$
\begin{matrix} 
t_{iu}=\frac{2d_{B}}{V_{1}}\cos i_{c}  & t_{id}=\frac{2d_{A}}{V_{1}}\cos i_{c} \\
d_{B}=\frac{t_{iu}V_{1}}{2\cos i_{c}} & d_{A}=\frac{t_{id}V_{1}}{2\cos i_{c}} \\
\boxed{ d_{B}=16.69\text{ m} } & \boxed{ d_{A}=33.38\text{ m} }
\end{matrix}
