$$
\phi(x,y)=\frac{x}{a}+\frac{y}{b}-1
$$
$$
G=\sqrt{ (x_{0}-x)^{2}+(y_{0}-y)^{2} }+\lambda(\frac{x}{a}+\frac{y}{b}-1)
$$
$$
\begin{cases}
\frac{ \partial G }{ \partial x } = \frac{x-x_{0}}{\sqrt{ (x_{0}-x)^{2}+(y_{0}-y)^{2} }}+\frac{\lambda}{a} =0 & \implies & \lambda=-\frac{a(x-x_{0})}{\sqrt{ (x_{0}-x)^{2}+(y_{0}-y)^{2} }} \\
\frac{ \partial G }{ \partial y } = \frac{y-y_{0}}{\sqrt{ (x_{0}-x)^{2}+(y_{0}-y)^{2} }}+\frac{\lambda}{b} =0 & \implies & \lambda=-\frac{b(y-y_{0})}{\sqrt{ (x_{0}-x)^{2}+(y_{0}-y)^{2} }}\\
\frac{ \partial G }{ \partial \lambda } = \frac{x}{a}+\frac{y}{b}-1 =0 
\end{cases}
$$
$$
\begin{matrix}
b(y-y_{0})=a(x-x_{0}) \\
y=\frac{a}{b}(x-x_{0})+y_{0} \\
\frac{bx}{a}+\frac{a}{b}(x-x_{0})+y_{0}-b =0  \\
\boxed{ x=\frac{a^{2}x_{0}-aby_{0}+ab^{2}}{a^{2}+b^{2}} } \\
\boxed{ y=\frac{-abx_{0}+b^{2}y_{0}+a^{2}b}{a^{2}+b^{2}} }
\end{matrix}
$$

$$
\begin{matrix} 
S(x,y,z)=2xy+2yz+2xz  &  & V(x,y,z)=xyz \\
\phi_{s}=2xy+2yz+2xz-s=0 &  & \phi_{v}=xyz-v=0
\end{matrix}
$$
## א
$$
G=2xy+2yz+2xz+\lambda(xyz-v)
$$
$$
\begin{cases}
\frac{ \partial G }{ \partial y } =2x+2z+\lambda xz =0  & \implies & \lambda=- \frac{2x+2z}{xz} \\
\frac{ \partial G }{ \partial x } =2y+2z+\lambda yz =0 & \implies & z=-\frac{2y}{2+\lambda y} \\
\frac{ \partial G }{ \partial z } =2x+2y+\lambda xy =0 & \implies & x=-\frac{2y}{2+\lambda y}=z \\
\frac{ \partial G }{ \partial \lambda } =xyz-v =0 \\
\end{cases}
$$
$$
\begin{matrix} 
\lambda= -\frac{2x+2x}{x^{2}}=-\frac{4}{x}  \\
\lambda=\frac{4(2+\lambda y)}{2y} \\
0=4+\lambda y \\
y=-\frac{4}{\lambda} \\
x=z=\frac{\frac{8}{\lambda}}{2-4}=-\frac{4}{\lambda} \\
\boxed{ x=y=z }
\end{matrix}
$$
## ב
$$
G=xyz+\lambda(2xy+2yz+2xz-s)
$$
$$
\begin{cases}
\frac{ \partial G }{ \partial x } = yz+2\lambda y+2\lambda z=0  & \implies & \lambda=-\frac{yz}{2y+2z} \\
\frac{ \partial G }{ \partial y } = xz+2\lambda x+2\lambda z=0  & \implies & z=-\frac{2\lambda x}{2\lambda+x}\\
\frac{ \partial G }{ \partial z } = xy+2\lambda x+2\lambda y=0 & \implies & y=-\frac{2\lambda x}{2\lambda+x}=z\\
\frac{ \partial G }{ \partial \lambda } = 2xy+2yz+2xz-s=0
\end{cases}
$$
$$
\begin{matrix}
\lambda=-\frac{y^{2}}{4y} \\
\lambda=\frac{2\lambda x}{8\lambda+4x} \\
x=-4\lambda \\
y=z=-\frac{8\lambda^{2}}{2\lambda}=-4\lambda \\
\boxed{ x=y=z }
\end{matrix}
$$

# 3
$$
J\left( \frac{x,y,z}{u,v,\phi} \right)=\begin{vmatrix}
\frac{ \partial x }{ \partial u }  & \frac{ \partial x }{ \partial v } & \frac{ \partial x }{ \partial \phi }   \\
\frac{ \partial y }{ \partial u }  & \frac{ \partial y }{ \partial v } & \frac{ \partial y }{ \partial \phi }   \\
\frac{ \partial z }{ \partial u }  & \frac{ \partial z }{ \partial v } & \frac{ \partial z }{ \partial \phi }  
\end{vmatrix}
=\begin{vmatrix}
v\cos \phi & u\cos \phi & -uv\sin \phi \\
v\sin \phi & u\sin \phi & uv\cos \phi \\
u & -v & 0
\end{vmatrix}
$$
$$
\begin{matrix} 
=u\begin{vmatrix}
u\cos \phi & -uv\sin \phi \\
u\sin \phi & uv\cos \phi
\end{vmatrix}+v \begin{vmatrix}
v\cos \phi & -uv\sin \phi \\
v\sin \phi & uv\cos \phi
\end{vmatrix} \\
=u[u^{2}v\cos ^{2}\phi+u^{2}v\sin ^{2}\phi]+v[uv^{2}\cos ^{2}\phi+uv^{2}\sin ^{2}\phi]  \\
=\boxed{ u^{3}v+uv^{3} }
\end{matrix}
$$
$$
\boxed{ J\left( \frac{u,v,\phi}{x,y,z} \right)=\frac{1}{u^{3}v+uv^{3}} }
$$
